Contenido completo de `ejercicio3/scripts/run_pack_c_v2.py`:

```python
"""Runner para Pack C v2 del Ej3.

Patrón: paralelismo OUTER sobre configs (cada cell = un config completo).
K=5 folds secuenciales adentro de cada worker (control con OMP=1 para
no contender). Persistencia por cell ANTES de devolver al master.

Uso:
    python -m ejercicio3.scripts.run_pack_c_v2 [--workers N]
"""
from __future__ import annotations

import os
os.environ.setdefault("OMP_NUM_THREADS", "1")
os.environ.setdefault("MKL_NUM_THREADS", "1")
os.environ.setdefault("OPENBLAS_NUM_THREADS", "1")
os.environ.setdefault("VECLIB_MAXIMUM_THREADS", "1")
os.environ.setdefault("NUMEXPR_NUM_THREADS", "1")

import argparse
import json
import sys
import time
import traceback
from concurrent.futures import ProcessPoolExecutor, as_completed
from datetime import datetime
from pathlib import Path

import pandas as pd

ROOT = Path(__file__).resolve().parent.parent.parent
sys.path.insert(0, str(ROOT))
from mlp.train import run_experiment  # noqa: E402

CSV_ROOT = ROOT

CONFIGS = [
    "ejercicio3/configs/base_extra_data_v2.json",
    "ejercicio3/configs/pack_c_v2/l2.json",
    "ejercicio3/configs/pack_c_v2/l2_aug.json",
    "ejercicio3/configs/pack_c_v2/dropout.json",
    "ejercicio3/configs/pack_c_v2/wider_l2_aug.json",
    "ejercicio3/configs/pack_c_v2/aug_s010.json",
]


def run_one_cell(cfg_path_str: str, out_root_str: str) -> dict:
    """Worker. Carga el config, corre run_experiment, persiste a disco."""
    cfg_path = Path(cfg_path_str)
    out_root = Path(out_root_str)
    cfg = json.loads(cfg_path.read_text())
    cell_id = cfg["model_name"]
    cell_dir = out_root / cell_id
    cell_dir.mkdir(parents=True, exist_ok=True)

    run_dir = run_experiment(cfg, csv_root=CSV_ROOT, output_dir=cell_dir, workers=1)

    s = pd.read_csv(run_dir / "run_summary.csv")
    s = s[~s["fold"].isin(["mean", "std"])].copy()
    s["fold"] = s["fold"].astype(int)
    s["cell_id"] = cell_id
    s["config_file"] = str(cfg_path.name)
    s.to_csv(cell_dir / "summary.csv", index=False)

    h = pd.read_csv(run_dir / "epoch_history.csv")
    h["cell_id"] = cell_id
    h.to_csv(cell_dir / "history.csv", index=False)

    return {"cell_id": cell_id, "run_dir": str(run_dir)}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--workers", type=int, default=8)
    parser.add_argument("--out-dir", type=Path, default=ROOT / "ejercicio3/output/v2")
    args = parser.parse_args()

    out_dir = args.out_dir
    out_dir.mkdir(parents=True, exist_ok=True)
    errors_log = out_dir / "errors.log"
    status_file = out_dir / "STATUS.txt"

    cfg_paths = [ROOT / c for c in CONFIGS]
    total = len(cfg_paths)

    t0 = time.time()
    print(f"[run_pack_c_v2] starting {total} cells with {args.workers} workers", flush=True)
    status_file.write_text(f"start: {datetime.now().isoformat()}\n0/{total} cells done\n")

    n_ok = 0
    n_fail = 0
    results = []

    with ProcessPoolExecutor(max_workers=args.workers) as ex:
        futures = {ex.submit(run_one_cell, str(p), str(out_dir)): p for p in cfg_paths}
        for fut in as_completed(futures):
            cfg_p = futures[fut]
            try:
                res = fut.result()
                results.append(res)
                n_ok += 1
                elapsed = time.time() - t0
                msg = f"[{n_ok+n_fail}/{total}] OK {res['cell_id']} ({elapsed:.0f}s)"
                print(msg, flush=True)
                status_file.write_text(
                    f"start: {datetime.now().isoformat()}\n"
                    f"{n_ok+n_fail}/{total} cells done ({n_ok} ok, {n_fail} fail)\n"
                    f"elapsed: {elapsed:.0f}s\n"
                )
            except Exception:
                n_fail += 1
                tb = traceback.format_exc()
                with errors_log.open("a") as f:
                    f.write(f"=== {cfg_p.name} ===\n{tb}\n\n")
                print(f"[{n_ok+n_fail}/{total}] FAIL {cfg_p.name} (see errors.log)", flush=True)

    elapsed = time.time() - t0
    print(f"\n[run_pack_c_v2] done: {n_ok}/{total} OK, {n_fail} fail, {elapsed:.0f}s total")
    status_file.write_text(
        f"done: {datetime.now().isoformat()}\n"
        f"{n_ok}/{total} OK, {n_fail} fail\n"
        f"elapsed: {elapsed:.0f}s\n"
    )

    return 0 if n_fail == 0 else 1


if __name__ == "__main__":
    sys.exit(main())
```

Contenido completo de `ejercicio3/analisis/plot_05_why_regularization.py`:

```python
"""Plot 05 — Por qué regularización (no arch ni HP).

Bar chart: para cada optimizer (con su LR óptimo del cross_v1), muestra
train_loss_final, val_loss_final, y la diferencia (gap = val - train).
Adam debería tener el gap más grande → justifica regularización como
palanca correcta.

Data source: cross_v1 stage 2 raw.csv (un subconjunto: arch_shallow
+ LR óptimo de cada opt + 3 seeds × 5 folds = 15 corridas por punto).

LR óptimo por opt (del análisis del cross_v1):
- sgd: 1e-2
- momentum: 1e-2
- adam: 1e-3
"""
from __future__ import annotations

import argparse
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd


OPT_LRS = {"sgd": 0.01, "momentum": 0.01, "adam": 0.001}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument(
        "--cross-csv",
        default="ejercicio2_experimentacion/output/cross_v1/stage2/raw.csv",
        type=Path,
    )
    parser.add_argument("--out", default="ejercicio3/presentacion/05_why_regularization.png", type=Path)
    args = parser.parse_args()

    df = pd.read_csv(args.cross_csv)

    rows = []
    for opt, lr in OPT_LRS.items():
        sub = df[(df["optimizer"] == opt) & (np.isclose(df["lr"], lr)) & (df["arch_name"] == "arch_shallow")]
        if sub.empty:
            raise RuntimeError(f"No data for {opt}@{lr} arch_shallow in {args.cross_csv}")
        rows.append({
            "opt": opt,
            "lr": lr,
            "train_loss": sub["train_loss_final"].mean(),
            "val_loss": sub["val_loss_final"].mean(),
            "gap": sub["val_loss_final"].mean() - sub["train_loss_final"].mean(),
            "n": len(sub),
        })
    agg = pd.DataFrame(rows)
    print(agg)

    opts = agg["opt"].tolist()
    x = np.arange(len(opts))
    width = 0.27

    fig, ax = plt.subplots(figsize=(10, 5.5))
    b1 = ax.bar(x - width, agg["train_loss"], width, label="train_loss_final (CE)", color="#4C9F70")
    b2 = ax.bar(x,         agg["val_loss"],   width, label="val_loss_final (CE)",   color="#E07A5F")
    b3 = ax.bar(x + width, agg["gap"],        width, label="gap = val − train",     color="#9D4EDD")

    for bars in [b1, b2, b3]:
        for bar in bars:
            ax.text(bar.get_x() + bar.get_width()/2, bar.get_height() + 0.005,
                    f"{bar.get_height():.3f}", ha="center", va="bottom", fontsize=9)

    ax.set_xticks(x)
    ax.set_xticklabels([f"{o}\n(LR={agg.iloc[i]['lr']:.0e})" for i, o in enumerate(opts)])
    ax.set_ylabel("Cross-entropy")
    ax.set_title("Train vs val loss y gap por optimizer — arch_shallow, mean sobre 3 seeds × 5 folds (cross_v1)")
    ax.legend(loc="upper left")
    ax.grid(True, axis="y", alpha=0.3)

    args.out.parent.mkdir(parents=True, exist_ok=True)
    fig.tight_layout()
    fig.savefig(args.out, dpi=130, bbox_inches="tight")
    print(f"Saved: {args.out}")


if __name__ == "__main__":
    main()
```

```python
"""Plot 07 — Scatter val_acc K=5 mean vs test_acc por config v2.

Línea identidad y=x. Si TODOS los puntos caen por debajo, hay un offset
sistemático val → test = distribution shift residual confirmado.
"""
from __future__ import annotations

import argparse
import json
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out-v2", default="ejercicio3/output/v2", type=Path)
    parser.add_argument("--out", default="ejercicio3/presentacion/07_val_vs_test_gap.png", type=Path)
    args = parser.parse_args()

    raw_s = pd.read_csv(args.out_v2 / "raw_summary.csv")
    val_agg = raw_s.groupby("cell_id").agg(
        val_acc_mean=("val_acc_final", "mean"),
        val_acc_sem=("val_acc_final", lambda x: x.std() / np.sqrt(len(x))),
    ).reset_index()

    test_rows = []
    for final_dir in sorted(args.out_v2.glob("final_eval_*/")):
        cfg = json.loads((final_dir / "config.json").read_text())
        cfg_name = cfg["model_name"]
        tm = pd.read_csv(final_dir / "test_metrics.csv").iloc[0]
        test_rows.append({"cell_id": cfg_name, "test_acc": float(tm["test_accuracy"])})
    test_df = pd.DataFrame(test_rows)

    merged = val_agg.merge(test_df, on="cell_id", how="inner")
    print(merged)

    fig, ax = plt.subplots(figsize=(8.5, 7))
    short_labels = {
        "base_extra_data_v2": "base",
        "pack_c_v2_l2": "l2",
        "pack_c_v2_l2_aug": "l2_aug",
        "pack_c_v2_dropout": "dropout",
        "pack_c_v2_wider_l2_aug": "wider",
        "pack_c_v2_aug_s010": "σ=0.10",
    }
    for _, r in merged.iterrows():
        ax.errorbar(r["val_acc_mean"], r["test_acc"], xerr=r["val_acc_sem"],
                    marker="o", markersize=10, capsize=4)
        ax.annotate(short_labels.get(r["cell_id"], r["cell_id"]),
                    (r["val_acc_mean"], r["test_acc"]),
                    xytext=(7, 7), textcoords="offset points", fontsize=10)

    lo = min(merged["val_acc_mean"].min(), merged["test_acc"].min()) - 0.005
    hi = max(merged["val_acc_mean"].max(), merged["test_acc"].max()) + 0.005
    ax.plot([lo, hi], [lo, hi], "k--", alpha=0.5, label="y = x (sin shift)")
    ax.set_xlabel("val_acc K=5 mean (entrenado con digits.csv + more_digits.csv)")
    ax.set_ylabel("test_acc (digits_test.csv)")
    ax.set_title(
        "Distribution shift residual: val_acc K=5 vs test_acc por config\n"
        "Todas las configs caen por debajo de y=x → offset sistemático ≈ shift residual"
    )
    ax.legend()
    ax.grid(True, alpha=0.3)
    ax.set_xlim(lo, hi)
    ax.set_ylim(lo, hi)
    ax.set_aspect("equal", adjustable="box")

    args.out.parent.mkdir(parents=True, exist_ok=True)
    fig.tight_layout()
    fig.savefig(args.out, dpi=130, bbox_inches="tight")
    print(f"Saved: {args.out}")

    mean_gap = (merged["val_acc_mean"] - merged["test_acc"]).mean()
    print(f"\nMean val→test gap: {mean_gap*100:.2f} pp")


if __name__ == "__main__":
    main()
```

```python
"""Plot 04 — Curvas de aprendizaje (train+val loss y acc) del ganador v2.

Mean ± std sobre 5 folds del K-fold del entrenamiento de validación
(no del final_eval, que tiene 1 sola corrida).
"""
from __future__ import annotations

import argparse
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--ganador-cell-id", required=True)
    parser.add_argument("--out-v2", default="ejercicio3/output/v2", type=Path)
    parser.add_argument("--out", default="ejercicio3/presentacion/04_curvas_aprendizaje_ej3.png", type=Path)
    args = parser.parse_args()

    history = pd.read_csv(args.out_v2 / args.ganador_cell_id / "history.csv")

    agg = history.groupby("epoch").agg(
        train_loss_mean=("train_loss", "mean"),
        train_loss_std=("train_loss", "std"),
        val_loss_mean=("val_loss", "mean"),
        val_loss_std=("val_loss", "std"),
        train_acc_mean=("train_acc", "mean"),
        train_acc_std=("train_acc", "std"),
        val_acc_mean=("val_acc", "mean"),
        val_acc_std=("val_acc", "std"),
    ).reset_index()

    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(13, 5))

    ax1.plot(agg["epoch"], agg["train_loss_mean"], label="train_loss", color="#4C9F70")
    ax1.fill_between(agg["epoch"], agg["train_loss_mean"]-agg["train_loss_std"], agg["train_loss_mean"]+agg["train_loss_std"], color="#4C9F70", alpha=0.2)
    ax1.plot(agg["epoch"], agg["val_loss_mean"], label="val_loss", color="#E07A5F")
    ax1.fill_between(agg["epoch"], agg["val_loss_mean"]-agg["val_loss_std"], agg["val_loss_mean"]+agg["val_loss_std"], color="#E07A5F", alpha=0.2)
    ax1.set_xlabel("Época"); ax1.set_ylabel("Cross-entropy")
    ax1.set_title("Loss")
    ax1.legend(); ax1.grid(True, alpha=0.3)

    ax2.plot(agg["epoch"], agg["train_acc_mean"], label="train_acc", color="#4C9F70")
    ax2.fill_between(agg["epoch"], agg["train_acc_mean"]-agg["train_acc_std"], agg["train_acc_mean"]+agg["train_acc_std"], color="#4C9F70", alpha=0.2)
    ax2.plot(agg["epoch"], agg["val_acc_mean"], label="val_acc", color="#E07A5F")
    ax2.fill_between(agg["epoch"], agg["val_acc_mean"]-agg["val_acc_std"], agg["val_acc_mean"]+agg["val_acc_std"], color="#E07A5F", alpha=0.2)
    ax2.set_xlabel("Época"); ax2.set_ylabel("Accuracy")
    ax2.set_title("Accuracy")
    ax2.legend(); ax2.grid(True, alpha=0.3)

    fig.suptitle(f"Curvas de aprendizaje (mean ± std sobre 5 folds) — {args.ganador_cell_id}", fontsize=12)
    fig.tight_layout(rect=[0, 0, 1, 0.96])

    args.out.parent.mkdir(parents=True, exist_ok=True)
    fig.savefig(args.out, dpi=130, bbox_inches="tight")
    print(f"Saved: {args.out}")


if __name__ == "__main__":
    main()
```
