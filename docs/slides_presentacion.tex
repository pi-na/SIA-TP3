% ============= 5. XOR interpretación =============
\begin{frame}{Ej 0 — Qué nos dice el resultado del XOR}
\textbf{Resultado clásico} (Minsky \& Papert, 1969 → Rumelhart \& Hinton, 1986):\\[0.4em]
\textit{El perceptrón simple no resuelve XOR; un MLP con una capa oculta no lineal sí.}\\[1em]

\textbf{Lo que la convergencia confirma:}
\begin{itemize}
  \item \textbf{Forward pass} de la red está bien implementado.
  \item \textbf{Backpropagation} computa los gradientes correctos (si no, el loss no bajaría).
  \item Las activaciones \textbf{tanh} y sus derivadas están bien definidas.
  \item El loss \textbf{MSE} y su gradiente también.
\end{itemize}
\vspace{0.6em}

\begin{takeaway}
Esta validación es la base para los ejercicios siguientes: la misma pipeline se reutiliza para clasificación de dígitos cambiando solo activación final (softmax) y loss (cross-entropy).
\end{takeaway}
\end{frame}

% ============= 8. MSE comparison =============
\begin{frame}{Ej 1 — Lineal vs No-Lineal: MSE de distillation}
\textbf{Mismo dataset, K-fold = 5 estratificado, mismas features.} Diferencia: presencia de la sigmoide.\\[0.8em]

\centering
\begin{tabular}{@{}lr@{}}
\toprule
\textbf{Modelo} & \textbf{MSE test (mean $\pm$ std)} \\
\midrule
Lineal (Adaline)        & $0.0266 \pm 0.0008$ \\
\rowcolor{accent2!15}
\textbf{No-Lineal (sigmoide)} & $\boldsymbol{0.0110 \pm 0.0004}$ \quad ($\boldsymbol{-59\%}$) \\
\bottomrule
\end{tabular}\\[1.2em]

\begin{insight}
\textbf{¿Por qué la sigmoide gana tan claro?}\\
El target es una probabilidad acotada en $[0,1]$. La sigmoide \emph{naturalmente} mapea su salida a ese rango. El lineal puede producir valores fuera de $[0,1]$ y el MSE penaliza fuerte esa salida fuera de dominio.
\end{insight}
\end{frame}

% ============= NUEVO. Mini-batch SGD =============
\begin{frame}{Ej 2 — Cómo se entrena: mini-batch SGD}
\textbf{Tres formas de aplicar updates en gradient descent:}\\[0.4em]

\centering\small
\begin{tabular}{@{}lcl@{}}
\toprule
\textbf{Modo} & \textbf{Update por} & \textbf{Trade-off} \\
\midrule
Online (SGD puro)    & 1 muestra      & Gradiente ruidoso, lento (sin vectorizar) \\
\rowcolor{accent2!15}
\textbf{Mini-batch}  & \textbf{B muestras} & \textbf{Estable + rápido (matmul de NumPy)} \\
Full-batch           & todo el dataset & Estable pero costoso por update \\
\bottomrule
\end{tabular}\\[0.5em]

\raggedright\small
\textbf{Pseudo-código por epoch} (lo que hace Ej 2 y Ej 3):
\textbf{1.}~shuffle del train set $\to$ \textbf{2.}~partir en chunks de $B$ muestras $\to$ \textbf{3.}~para cada chunk: forward + backward + un único update.\\[0.5em]

\normalsize
\begin{insight}
Ej 1 (perceptrón simple) usa \textbf{online estricto} siguiendo la regla del apunte: un update por muestra. Ej 2 y Ej 3 usan \textbf{mini-batch} con $B$ determinado por sweep (Fase 1.4).
\end{insight}
\end{frame}

% ============= 25. Qué movió la aguja =============
\begin{frame}{Ej 3 — Análisis: qué movió la aguja, qué no}
\begin{insight}
\textbf{Lo que ayudó:}
\begin{itemize}
  \item \textbf{Más datos:} \textbf{+10 pp.} El movimiento dominante.
  \item \textbf{L2:} $+0.20$ pp consistente.
  \item \textbf{Aug gaussiana:} $+0.32$ pp \emph{adicional} sobre L2 (sinergia).
\end{itemize}
\end{insight}

\begin{warning}
\textbf{Lo que no ayudó:}
\begin{itemize}\setlength\itemsep{0.1em}
  \item \textbf{Dropout:} gana en val, pierde en test. Reduce varianza al fold, no al shift.
  \item \textbf{wider} \([784,200,100,10]\): mejor en val, peor en test. Descarta falta de capacidad.
  \item \textbf{Combinar todo:} L2+dropout+aug peor que L2+aug solo.
\end{itemize}
\end{warning}
\end{frame}

% ============= 26. Por qué no llegamos al 98 =============
\begin{frame}{Ej 3 — Por qué no llegamos al 98\%}
\centering
\Large \textcolor{warn}{\textbf{Mejor: 96.88\%. Faltan 1.12 pp.}}\\[0.6em]

\normalsize\raggedright
\textbf{Tres hipótesis ordenadas por probabilidad:}
\begin{enumerate}\setlength\itemsep{0.2em}
  \item \textbf{Augmentation insuficiente.} Gaussian noise es \emph{isotrópico} (por píxel). El shift parece ser de \emph{estilo de escritura}: rotación, traslación, grosor de trazo. Pide augmentation \textbf{geométrica}, no implementada acá.
  \item \textbf{Sub-representación en val interno del final\_eval.} Val 10\% random para early stopping puede no incluir los digits difíciles del test.
  \item \textbf{Capacidad}: improbable, \code{wider} sobreajustó.
\end{enumerate}
\vspace{0.4em}

\begin{takeaway}
\textbf{Próximo paso}: augmentation por rotación/traslación pequeñas.
\end{takeaway}
\end{frame}
